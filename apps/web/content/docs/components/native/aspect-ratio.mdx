## Installation

```bash
npx fabricator-ui@latest add aspect-ratio
```

## Usage

```tsx
import { AspectRatio } from "@/components/ui/aspect-ratio"
import { Text } from "@/components/ui/text"
```

```tsx
<AspectRatio
  ratio={16 / 9}
  className="overflow-hidden rounded-lg bg-muted"
>
  <Image
    src={PHOTO}
    alt="An office with plants by the window"
    className="size-full"
  />
</AspectRatio>
```

## Examples

```tsx showLineNumbers title="aspect-ratio-example.tsx"
import { Image } from "react-native"

import { AspectRatio } from "@/components/ui/aspect-ratio"
import { Text } from "@/components/ui/text"

const PHOTO =
  "https://images.unsplash.com/photo-1497366754035-f200968a6e72?w=900&auto=format&fit=crop&q=80"

function AspectRatioExample() {
  return (
    <ExampleWrapper>
      <Example title="16:9" className="items-stretch">
        <AspectRatio
          ratio={16 / 9}
          className="overflow-hidden rounded-lg bg-muted"
        >
          <Image
            src={PHOTO}
            alt="An office with plants by the window"
            className="size-full"
          />
        </AspectRatio>
      </Example>
      <Example title="Square" className="items-stretch">
        <AspectRatio
          ratio={1}
          className="items-center justify-center rounded-lg bg-muted md:w-60"
        >
          <Text className="text-muted-foreground">1:1</Text>
        </AspectRatio>
      </Example>
      <Example title="Portrait" className="items-stretch">
        <AspectRatio
          ratio={9 / 16}
          className="w-1/2 items-center justify-center rounded-lg bg-muted"
        >
          <Text className="text-muted-foreground">9:16</Text>
        </AspectRatio>
      </Example>
    </ExampleWrapper>
  )
}

export { AspectRatioExample }
```

## API reference



Props it adds to what it wraps: `ratio`.

## Differences from web

**Same.** It works as it does on the web, with `onPress` for `onClick`.

## Accessibility

Text is read as it appears. Give a control with no text an `accessibilityLabel`.
