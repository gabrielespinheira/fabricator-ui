## Installation

```bash
npx fabricator-ui@latest add kbd
```

## Usage

```tsx
import { Button } from "@/components/ui/button"
import { Kbd, KbdGroup } from "@/components/ui/kbd"
```

```tsx
<Kbd>Esc</Kbd>
<Kbd>K</Kbd>
<Kbd>
  <CommandIcon />
</Kbd>
```

## Composition

- `Kbd`
- `KbdGroup`

## Examples

```tsx showLineNumbers title="kbd-example.tsx"
import { CommandIcon } from "@fabricator-ui/icons-react-native"

import { Button } from "@/components/ui/button"
import { Kbd, KbdGroup } from "@/components/ui/kbd"

function KbdExample() {
  return (
    <ExampleWrapper>
      <Example title="Keys" className="flex-row flex-wrap items-center">
        <Kbd>Esc</Kbd>
        <Kbd>K</Kbd>
        <Kbd>
          <CommandIcon />
        </Kbd>
      </Example>
      <Example title="Group">
        <KbdGroup>
          <Kbd>
            <CommandIcon />
          </Kbd>
          +<Kbd>B</Kbd>
        </KbdGroup>
      </Example>
      <Example title="In a button">
        <Button variant="outline">
          Accept
          <Kbd>⏎</Kbd>
        </Button>
      </Example>
    </ExampleWrapper>
  )
}

export { KbdExample }
```

## Differences from web

**Same.** Visual only. Useful on tablets with a hardware keyboard. It doesn't invert inside a tooltip.

## Accessibility

Text is read as it appears. Give a control with no text an `accessibilityLabel`.
