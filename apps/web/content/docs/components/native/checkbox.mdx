## Installation

```bash
npx fabricator-ui@latest add checkbox
```

## Usage

```tsx
import { Checkbox } from "@/components/ui/checkbox"
import { Text } from "@/components/ui/text"
```

```tsx
<Checkbox accessibilityLabel="Off" />
<Checkbox accessibilityLabel="On" defaultChecked />
<Checkbox accessibilityLabel="Mixed" indeterminate />
<Checkbox accessibilityLabel="Invalid" aria-invalid />
<Checkbox accessibilityLabel="Disabled" disabled />
<Checkbox accessibilityLabel="Disabled on" disabled defaultChecked />
```

## Examples

```tsx showLineNumbers title="checkbox-example.tsx"
import * as React from "react"
import { Pressable, View } from "react-native"

import { Checkbox } from "@/components/ui/checkbox"
import { Text } from "@/components/ui/text"

function CheckboxExample() {
  const [isChecked, setIsChecked] = React.useState(true)
  return (
    <ExampleWrapper>
      <Example title="States" className="flex-row items-center">
        <Checkbox accessibilityLabel="Off" />
        <Checkbox accessibilityLabel="On" defaultChecked />
        <Checkbox accessibilityLabel="Mixed" indeterminate />
        <Checkbox accessibilityLabel="Invalid" aria-invalid />
        <Checkbox accessibilityLabel="Disabled" disabled />
        <Checkbox accessibilityLabel="Disabled on" disabled defaultChecked />
      </Example>
      <Example title="With a label">
        <Pressable
          className="flex-row items-center gap-2"
          onPress={() => setIsChecked(!isChecked)}
        >
          <Checkbox
            accessibilityLabel="Accept the terms"
            checked={isChecked}
            onCheckedChange={setIsChecked}
          />
          <Text>Accept the terms</Text>
        </Pressable>
        <View>
          <Text className="text-muted-foreground">
            {isChecked ? "Accepted" : "Not accepted"}
          </Text>
        </View>
      </Example>
    </ExampleWrapper>
  )
}

export { CheckboxExample }
```

## API reference



Props it adds to what it wraps: `"aria-invalid"`, `checked`, `defaultChecked`, `indeterminate`, `onCheckedChange`.

## Differences from web

**Same.** It works as it does on the web, with `onPress` for `onClick`.

## Accessibility

Parts set `role="checkbox"`, read by VoiceOver and TalkBack. Give a control with no text an `accessibilityLabel`.
