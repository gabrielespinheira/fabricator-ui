---
title: Folder
description: A folder as a tile, the way a drive shows one, with its first files peeking out. Hover lifts them out and tips the front open.
base: native
component: true
---

## Installation

```bash
npx fabricator-ui@latest add folder
```

## Usage

```tsx
import { Badge } from "@/components/ui/badge"
import {
  Folder,
  FolderCover,
  FolderDescription,
  FolderFile,
  FolderFiles,
  FolderPreview,
  FolderTitle,
} from "@/components/ui/folder"
import { Text } from "@/components/ui/text"
```

```tsx
<View className="flex-row flex-wrap justify-center">
  <Folder
    variant="glass"
    className="w-1/2"
    onPress={() => setOpened("Onboarding")}
  >
    <FolderPreview>
      <Sheets count={2} />
      <FolderCover>
        <Text>MA</Text>
        <Badge variant="secondary">Updated 2h</Badge>
      </FolderCover>
    </FolderPreview>
    <FolderTitle>Onboarding</FolderTitle>
    <FolderDescription>15 files · 3 folders</FolderDescription>
  </Folder>
  <Folder
    color="blue"
    className="w-1/2"
    onPress={() => setOpened("Contracts")}
  >
    <FolderPreview>
      <Sheets count={3} />
      <FolderCover className="justify-end">
        <LockIcon accessibilityLabel="Private" />
      </FolderCover>
    </FolderPreview>
    <FolderTitle>Contracts</FolderTitle>
    <FolderDescription>12 files</FolderDescription>
  </Folder>
</View>
<Text className="text-center text-(length:--density-text-13) text-muted-foreground">
  {opened ? `Opened ${opened}` : "Nothing opened yet"}
</Text>
```

## Examples

import * as React from "react"
import { LockIcon } from "@fabricator-ui/icons-react-native"
import { View } from "react-native"

import { Badge } from "@/components/ui/badge"
import {
  Folder,
  FolderCover,
  FolderDescription,
  FolderFile,
  FolderFiles,
  FolderPreview,
  FolderTitle,
} from "@/components/ui/folder"
import { Text } from "@/components/ui/text"

const COLORS = [
  { color: undefined, name: "Neutral" },
  { color: "primary", name: "Primary" },
  { color: "blue", name: "Blue" },
  { color: "green", name: "Green" },
  { color: "orange", name: "Orange" },
  { color: "pink", name: "Pink" },

function Sheets({ count }: { count: number }) {
  return (
    <FolderFiles>
      {Array.from({ length: count }, (_, index) => (
        <FolderFile key={index} />
      ))}
    </FolderFiles>
  )

function FolderExample() {
  const [opened, setOpened] = React.useState<string | null>(null)
  return (
    <ExampleWrapper>
      <Example title="Tap to open">
        <View className="flex-row flex-wrap justify-center">
          <Folder
            variant="glass"
            className="w-1/2"
            onPress={() => setOpened("Onboarding")}
          >
            <FolderPreview>
              <Sheets count={2} />
              <FolderCover>
                <Text>MA</Text>
                <Badge variant="secondary">Updated 2h</Badge>
              </FolderCover>
            </FolderPreview>
            <FolderTitle>Onboarding</FolderTitle>
            <FolderDescription>15 files · 3 folders</FolderDescription>
          </Folder>
          <Folder
            color="blue"
            className="w-1/2"
            onPress={() => setOpened("Contracts")}
          >
            <FolderPreview>
              <Sheets count={3} />
              <FolderCover className="justify-end">
                <LockIcon accessibilityLabel="Private" />
              </FolderCover>
            </FolderPreview>
            <FolderTitle>Contracts</FolderTitle>
            <FolderDescription>12 files</FolderDescription>
          </Folder>
        </View>
        <Text className="text-center text-(length:--density-text-13) text-muted-foreground">
          {opened ? `Opened ${opened}` : "Nothing opened yet"}
        </Text>
      </Example>
      <Example title="Colors">
        <View className="flex-row flex-wrap justify-center">
          {COLORS.map(({ color, name }) => (
            <Folder key={name} color={color} size="sm" className="w-1/3">

                <Sheets count={2} />

              <FolderTitle>{name}</FolderTitle>
            </Folder>
          ))}
        </View>
      </Example>
      <Example title="Glass">
        <View className="flex-row flex-wrap justify-center">
          <Folder variant="glass" color="purple" className="w-1/2">
            <FolderPreview>
              <Sheets count={3} />

            <FolderTitle>Certificates</FolderTitle>
          </Folder>
        </View>
      </Example>
      <Example title="Empty and file counts">
        <View className="flex-row flex-wrap justify-center">
          {[0, 1, 2, 3, 12].map((count) => (
            <Folder key={count} color="cyan" size="sm" className="w-1/3">
              <FolderPreview>
                <Sheets count={count} />
              </FolderPreview>
              <FolderTitle>
                {count === 0
                  ? "Empty"
                  : `${count} ${count === 1 ? "file" : "files"}`}

          ))}

      </Example>
      <Example title="Sizes">

## API reference

| Prop | Values |
| --- | --- |
| `variant` | `default`, `glass` |
| `size` | `default`, `lg`, `sm` |

## Differences from web

**Touch input.** A press, or an iPad pointer resting on it, lifts the files out and tips the front open; `open` holds it, as hover and focus do on the website. `onPress` makes the tile a button. `color` takes a palette name, a role, hex, rgb(), hsl() or oklch(); the website also reads named CSS colours. Glass has no blur behind it, so its front is denser to frost the files. Text on the cover is the text colour rather than the folder colour's text token, and badges on it keep their own look instead of turning frosted pills. `size` sets the tile's width; its text follows the device's density.

## Accessibility

Text is read as it appears. Give a control with no text an `accessibilityLabel`.
