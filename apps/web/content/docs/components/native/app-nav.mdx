## Installation

```bash
npx fabricator-ui@latest add app-nav
```

## Usage

```tsx
import {
  AppNav,
  AppNavBottomBar,
  AppNavCustomize,
  type AppNavGroup,
  type AppNavItem,
} from "@/components/ui/app-nav"
import { AppShell, AppShellMain } from "@/components/ui/app-shell"
import { Button } from "@/components/ui/button"
import {
  Sidebar,
  SidebarContent,
  SidebarFooter,
  SidebarInset,
  SidebarProvider,
} from "@/components/ui/sidebar"
import { Text } from "@/components/ui/text"
```

```tsx
<Frame>
  <SidebarProvider isMobile={false}>
    <Sidebar collapsible="none">
      <SidebarContent>
        <AppNav
          groups={GROUPS}
          pathname={pathname}
          onNavigate={navigate}
          preferences={preferences}
          onPreferencesChange={setPreferences}
          customizable
          pinnedLabel="Pinned"
        />
      </SidebarContent>
      <SidebarFooter>
        <AppNavCustomize
          groups={GROUPS}
          preferences={preferences}
          onPreferencesChange={setPreferences}
          label="Customize"
          title="Customize sidebar"
          description="Choose what shows, and hold a grip to drag rows."
        />
        <AppNavCustomize
          groups={GROUPS}
          preferences={preferences}
          onPreferencesChange={setPreferences}
        >
          <Button variant="outline" />
        </AppNavCustomize>
      </SidebarFooter>
    </Sidebar>
    <Status item={pathname} />
  </SidebarProvider>
</Frame>
```

## API reference



Props it adds to what it wraps: `activeId`, `customizable`, `defaultPreferences`, `description`, `groups`, `label`, `onNavigate`, `onPreferencesChange`, `pathname`, `pinnedLabel`, `preferences`, `renderLink`, `title`.

## Differences from web

**Touch input.** Rows reorder on a long press, so a reorderable row's menu sits on its trailing button, and on the icon rail it has none. Keyboard reordering has no equivalent.

## Accessibility

Parts set `role="button"`, `role="heading"`, `role="navigation"`, `role="none"`, read by VoiceOver and TalkBack. Give a control with no text an `accessibilityLabel`.
