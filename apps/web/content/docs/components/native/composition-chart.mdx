---
title: Composition Chart
description: How the mix of a whole changes over time, as 100% columns with each part's current share.
base: native
component: true
---

## Installation

```bash
npx fabricator-ui@latest add composition-chart
```

## Usage

```tsx
import { type ChartConfig } from "@/components/ui/chart-kit"
import { CompositionChart } from "@/components/ui/composition-chart"
import { Text } from "@/components/ui/text"
```

```tsx
<CompositionChart
  data={traffic}
  index="date"
  config={trafficConfig}
  label="Share of visits by source"
  className="h-64 w-full"
/>
```

## Composition

- `CompositionChart`
- `CompositionChartProps`

function FollowingTheScrub() {
  const [month, setMonth] = React.useState<Plans | null>(null)
  const shown = month ?? plans[plans.length - 1]
  return (
    <>
      <Text className="text-sm text-muted-foreground">
        Paid accounts in {shown.month}:{" "}
        {(shown.pro + shown.team + shown.enterprise).toLocaleString("en-US")}
      </Text>
      <CompositionChart
        data={plans}
        index="month"
        config={plansConfig}
        label="Accounts by plan"
        onActiveChange={(active) => setMonth(active?.datum ?? null)}
        className="h-56 w-full"
      />
    </>
  )
}
