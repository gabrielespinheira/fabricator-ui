---
title: Accordion
description: A vertically stacked set of interactive headings that each reveal a section of content.
base: native
component: true
---

## Installation

```bash
npx fabricator-ui@latest add accordion
```

## Usage

```tsx
import {
  Accordion,
  AccordionContent,
  AccordionItem,
  AccordionTrigger,
} from "@/components/ui/accordion"
import { Text } from "@/components/ui/text"
```

```tsx
<Accordion defaultValue={["notifications"]}>
  <AccordionRows />
</Accordion>
```

## Composition

- `Accordion`
- `AccordionContent`
- `AccordionItem`
- `AccordionTrigger`

## Examples

```tsx showLineNumbers title="accordion-example.tsx"
import * as React from "react"

import {
  Accordion,
  AccordionContent,
  AccordionItem,
  AccordionTrigger,
} from "@/components/ui/accordion"
import { Text } from "@/components/ui/text"

const ITEMS = [
  {
    value: "notifications",
    trigger: "Notification settings",
    content: "Choose how you hear about updates: by email or by push.",
  },
  {
    value: "privacy",
    trigger: "Privacy and security",
    content:
      "Turn on two-factor authentication, manage connected devices and download your data.",
  },
  {
    value: "billing",
    trigger: "Billing and subscription",
    content: "See your plan, past invoices and the payment method on file.",
  },
]

function AccordionRows() {
  return ITEMS.map((item) => (
    <AccordionItem key={item.value} value={item.value}>
      <AccordionTrigger>{item.trigger}</AccordionTrigger>
      <AccordionContent>{item.content}</AccordionContent>
    </AccordionItem>
  ))
}

function AccordionExample() {
  const [value, setValue] = React.useState(["privacy"])
  return (
    <ExampleWrapper>
      <Example title="One open at a time">
        <Accordion defaultValue={["notifications"]}>
          <AccordionRows />
        </Accordion>
      </Example>
      <Example title="Several open">
        <Accordion multiple defaultValue={["notifications", "billing"]}>
          <AccordionRows />
        </Accordion>
      </Example>
      <Example title="Controlled">
        <Accordion value={value} onValueChange={setValue}>
          <AccordionRows />
        </Accordion>
        <Text className="text-muted-foreground">
          Open: {value.join(", ") || "nothing"}
        </Text>
      </Example>
      <Example title="A disabled item">
        <Accordion defaultValue={["available"]}>
          <AccordionItem value="available">
            <AccordionTrigger>Available</AccordionTrigger>
            <AccordionContent>
              You can open and close this one.
            </AccordionContent>
          </AccordionItem>
          <AccordionItem value="locked" disabled>
            <AccordionTrigger>Locked</AccordionTrigger>
            <AccordionContent>This one stays shut.</AccordionContent>
          </AccordionItem>
        </Accordion>
      </Example>
      <Example title="All disabled">
        <Accordion disabled>
          <AccordionRows />
        </Accordion>
      </Example>
    </ExampleWrapper>
  )
}

export { AccordionExample }
```

## API reference

Props it adds to what it wraps: `defaultValue`, `disabled`, `multiple`, `onValueChange`, `value`.

## Differences from web

**Same.** It works as it does on the web, with `onPress` for `onClick`.

## Accessibility

Text is read as it appears. Give a control with no text an `accessibilityLabel`.
